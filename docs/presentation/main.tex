% ── Slide 2 · Problem & Objective ───────────────────────
\begin{frame}{Problem \& Objective}
  \begin{itemize}
    \setlength\itemsep{0.75em}
    \item Small LLMs ($<$2B parameters) frequently fail to produce valid,
          schema-conformant JSON
    \item Common errors: unclosed brackets, type violations, conversational filler
    \item \textbf{Goal:} use GRPO with rule-based rewards to reach
          $>$90\% Pass@1 on strict JSON
    \item No neural reward model $\rightarrow$
          dense, \alert{deterministic}, interpretable signal
  \end{itemize}
\end{frame}

% ── Slide 4 · GRPO Algorithm ────────────────────────────
\begin{frame}{Group Relative Policy Optimization (GRPO)}
  \vspace{0.15em}
  \centering
  \resizebox{\textwidth}{!}{%
  \begin{tikzpicture}[
    node distance=0cm and 0.52cm,
    box/.style={
      rectangle, rounded corners=5pt,
      minimum width=2.5cm, minimum height=1.15cm,
      text centered, inner sep=6pt, text width=2.4cm,
      draw=navy, thick
    },
    arr/.style={-{Stealth[length=6pt]}, thick, color=navy},
  ]
    \node[box, fill=sky!15]  (p) {\small Prompt $x$\\[3pt]\tiny sampled from dataset};
    \node[box, fill=navy!10, right=0.6cm of p] (g) {\small $G{=}8$ completions\\[3pt]\tiny policy $\pi_\theta$, temp $T$};
    \node[box, fill=sky!28,  right=0.6cm of g] (r) {\small Reward $r_i$\\[3pt]\tiny 7 rule-based\\[-2pt]\tiny components};
    \node[box, fill=navy!15, right=0.6cm of r] (n) {\small Advantage $\hat{A}_i$\\[3pt]\tiny $(r_i - \mu_G)\,/\,\sigma_G$};
    \node[box, fill=sky!18,  right=0.6cm of n] (u) {\small Update $\pi_\theta$\\[3pt]\tiny gradient step\\[-2pt]\tiny KL reg.\ $\beta{=}0.04$};
    \draw[arr] (p) -- (g);
    \draw[arr] (g) -- (r);
    \draw[arr] (r) -- (n);
    \draw[arr] (n) -- (u);
    \draw[arr, dashed, color=slate!80, rounded corners=6pt]
      (u.south) -- ++(0,-0.65)
      -- node[below, font=\tiny, color=slate, inner sep=2pt]
         {updated policy becomes new reference for the next iteration}
      ($(p.south)+(0,-0.65)$) -- (p.south);
  \end{tikzpicture}}

  \vspace{0.6em}
  \raggedright
  \begin{itemize}
    \setlength\itemsep{0.42em}
    \item No neural reward model — 7 \alert{rule-based} components score each completion
    \item Additive rewards prevent \alert{zero-advantage collapse}
    \item 4-bit NF4 $+$ LoRA ($r{=}16$) on a single NVIDIA L40S GPU
  \end{itemize}
\end{frame}

% ── Slide 5 · Reward Functions ──────────────────────────
\begin{frame}{Rule-Based Reward System}
  \centering
  \renewcommand{\arraystretch}{1.38}
  \setlength\tabcolsep{8pt}
  \begin{tabularx}{0.94\linewidth}{@{}lcX@{}}
    \toprule
    \textbf{Component} & \textbf{Range} & \textbf{What it measures} \\
    \midrule
    Format     & $[0,\,1]$     & \texttt{```json```} code fence present \\
    Validity   & $[0,\,1]$     & JSON parseability (partial credit by error position) \\
    Schema     & $[0,\,1]$     & Structural conformance: keys, types, nesting depth \\
    Reasoning  & $[-0.5,\,1]$ & Quality of \texttt{<think>} block — Think mode only \\
    Truncation & $[-1,\,1]$   & Unclosed brackets / trailing commas \\
    Repetition & $[-1,\,1]$   & Degenerate loops, duplicate lines \\
    Strictness & $[-1,\,1]$   & Extra text outside the JSON block \\
    \bottomrule
  \end{tabularx}
  \vspace{0.5em}

  \small\textit{Additive score, not gating — gradient is always non-zero.}
\end{frame}

% ── Slide 10 · Think vs. No-Think ──────────────────────
\begin{frame}{Think Mode — Reasoning Tokens in Practice}
  \begin{columns}[T]
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \setlength\itemsep{0.45em}
        \item \alert{Think baseline is lower} for small models — the\\*
              \texttt{<think>} prompt disrupts their fragile context\\*
              \small SmolLM2-360M: 77.67\% $\to$ 69.33\% at baseline alone
        \item Think vs.\ No-Think gain (standard training):\\*
          \begin{tabular}{@{}lr@{}}
            SmolLM2-135M  & \alert{$+$2.33 pp} \\
            SmolLM2-360M  & \alert{$-$1.00 pp} \\
            Qwen2.5-0.5B  & $+$0.34 pp \\
            TinyLlama-1.1B & $+$1.00 pp \\
            Gemma-2-2B    & \alert{$-$0.67 pp} \\
          \end{tabular}
        \item Extra tokens $\rightarrow$ higher latency, negligible accuracy benefit
      \end{itemize}
      \vspace{0.3em}
      \begin{block}{Notable anomaly — Gemma-2-2B Think/Curriculum}
        \small U-shaped curve: Stage\,1 \textbf{83\%} $\to$ Stage\,2 \textbf{66\%}
        $\to$ Stage\,3 \textbf{97.67\%}\\*
        Forcing reasoning on simple schemas temporarily\\* collapses an already-aligned model
      \end{block}
    \end{column}

% ── Slide 11 · Lessons & Limitations ───────────────────
\begin{frame}{Lessons \& Limitations}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \textbf{Lessons}
      \begin{itemize}
        \setlength\itemsep{0.5em}
        \item Additive rewards $>$ gating\\*
              (prevent zero-advantage collapse)
        \item Standard training is \alert{sufficient} for weak models
        \item Curriculum is a \alert{safety net} for strong models,\\*
              not a universal performance booster
        \item Think mode adds cost but \alert{minimal accuracy gain}\\*
              for models $<$1B parameters
        \item 4-bit LoRA on a single GPU is sufficient
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \textbf{Limitations}
      \begin{itemize}
        \setlength\itemsep{0.5em}
        \item Evaluated on JSON only\\
              (not directly generalizable)
        \item Synthetic dataset $\rightarrow$\\
              potential real-world gap
        \item Single GPU, 2\,500 steps only
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}

% ── Slide 12 · Conclusions ──────────────────────────────
\begin{frame}{Conclusions}
  \begin{itemize}
    \setlength\itemsep{0.72em}
    \item GRPO $+$ rule-based rewards brings all 5 models to
          \alert{87–99\% Pass@1} on strict JSON generation
    \item \textbf{Standard training} is the go-to strategy for weak models —
          large gains with no curriculum overhead
    \item \textbf{Curriculum} is a safety net for strong models:\\*
          prevents the regression that standard training causes on Gemma-2-2B
    \item \textbf{Think mode} yields \alert{marginal gains} not worth the latency cost —
          small models cannot generate meaningful reasoning tokens
  \end{itemize}
\end{frame}
