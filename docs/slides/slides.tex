\begin{frame}{GRPO: Group-Relative Advantage Eliminates the Critic}
  \centering
  \resizebox{\textwidth}{!}{%
  \begin{tikzpicture}[
    node distance=0cm and 0.8cm,
    box/.style={
      rectangle, rounded corners=8pt,
      minimum width=3.3cm, minimum height=1.7cm,
      text centered, inner sep=7pt, text width=3.1cm,
      draw=bordergray!80, line width=0.7pt,
      top color=cardbg, bottom color=darkbg,
      text=textprimary,
      blur shadow={shadow xshift=2pt, shadow yshift=-2pt,
                   shadow blur steps=6, shadow blur radius=3pt,
                   shadow opacity=40},
    },
    sbox/.style={
      rectangle, rounded corners=5pt,
      minimum width=1.7cm, minimum height=0.75cm,
      text centered, inner sep=3pt, text width=1.5cm,
      draw=bordergray!60, line width=0.5pt,
      fill=cardbg, font=\scriptsize,
      text=textprimary,
    },
    arr/.style={-{Stealth[length=8pt, width=5pt]}, line width=1.4pt,
                color=sky!80, shorten >=2pt, shorten <=2pt},
    sarr/.style={-{Stealth[length=5pt]}, thick, color=sky!50},
    lbl/.style={font=\tiny\bfseries, color=textsecondary, fill=darkbg,
                inner sep=2pt, rounded corners=2pt},
  ]
    % ── Main pipeline nodes ──────────────────────────────
    \node[box] (p)
      {\small\textbf{Prompt} $x$\\[4pt]
       \scriptsize sampled from dataset\\
       difficulty $d\in\{S,M,H\}$};

    \node[box, right=0.8cm of p] (g)
      {\small\textbf{Sample} $G{=}8$\\[4pt]
       \scriptsize $y_1,\dots,y_8\sim\pi_\theta(\cdot|x)$\\
       temp $T\in[0.6,\,0.8]$};

    \node[box, right=0.8cm of g] (r)
      {\small\textbf{Score} $r_i$\\[4pt]
       \scriptsize $r_i=\sum\nolimits_k w_k\,r_k(y_i)$\\
       7 additive components};

    \node[box, right=0.8cm of r] (a)
      {\small\textbf{Advantage} $\hat{A}_i$\\[4pt]
       \scriptsize $\hat{A}_i=\dfrac{r_i-\bar{r}_G}{\sigma_{r_G}}$\\[2pt]
       group-relative norm.};

    \node[box, right=0.8cm of a] (u)
      {\small\textbf{Update} $\pi_\theta$\\[4pt]
       \scriptsize $\mathcal{L}=\sum_i\hat{A}_i\log\pi_\theta(y_i|x)$\\
       $-\,\beta\,D_{\mathrm{KL}}$,\quad$\beta{=}0.04$};

    % ── Glow dots at node headers ────────────────────────
    \foreach \nd/\col in {p/sky, g/purple, r/gold, a/accent, u/sky}{
      \fill[\col, opacity=0.12] (\nd.north) circle (8pt);
    }

    % ── Arrow labels ─────────────────────────────────────
    \draw[arr] (p) -- node[lbl, above]{prompt} (g);
    \draw[arr] (g) -- node[lbl, above]{$G$ completions} (r);
    \draw[arr] (r) -- node[lbl, above]{$\{r_i\}$} (a);
    \draw[arr] (a) -- node[lbl, above]{$\{\hat{A}_i\}$} (u);

    % ── Feedback loop ─────────────────────────────────────
    \draw[arr, densely dashed, color=textsecondary!60, rounded corners=12pt]
      (u.south) -- ++(0,-0.9)
      -- node[below, font=\tiny\itshape, color=textsecondary,
              fill=darkbg, inner sep=2pt, rounded corners=2pt, pos=0.5]
         {updated $\pi_\theta$ becomes new reference}
      ($(p.south)+(0,-0.9)$) -- (p.south);

    % ── Reward component sub-boxes (below Score node) ─────
    \node[sbox, below=1.3cm of r, xshift=-3.8cm] (c1)
      {\textcolor{sky}{\faCode}\\[-1pt]format\\$w{=}0.20$};
    \node[sbox, right=0.18cm of c1] (c2)
      {\textcolor{accent}{\faCheck}\\[-1pt]validity\\$w{=}0.25$};
    \node[sbox, right=0.18cm of c2] (c3)
      {\textcolor{gold}{\faSitemap}\\[-1pt]schema\\$w{=}0.25$};
    \node[sbox, right=0.18cm of c3] (c4)
      {\textcolor{warn}{\faTimes}\\[-1pt]trunc.\\$w{=}0.10$};
    \node[sbox, right=0.18cm of c4] (c5)
      {\textcolor{textsecondary}{\faRedo}\\[-1pt]repet.\\$w{=}0.10$};
    \node[sbox, right=0.18cm of c5] (c6)
      {\textcolor{purple}{\faBan}\\[-1pt]strict.\\$w{=}0.10$};

    % horizontal collector bar + single arrow to Score node
    \coordinate (left)  at (c1.north west);
    \coordinate (right) at (c6.north east);
    \coordinate (mid)   at ($(left)!0.5!(right)$);
    \draw[sarr, rounded corners=3pt]
      (left) -- (right)
      (mid)  -- (r.south);

  \end{tikzpicture}}
  \vspace{0.4em}

  \begin{tcolorbox}[insightbox=sky, width=0.88\textwidth]
    \small\textbf{Key insight} \textcolor{textsecondary}{|} No critic network needed. Advantages are computed \emph{within} each group of $G{=}8$ completions.
    Additive reward ensures gradient $\neq 0$ — avoids \alert{zero-advantage collapse}.
  \end{tcolorbox}
\end{frame}

\begin{frame}{Reward Design Matters More Than Model Size}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{tcolorbox}[insightbox=accent, title={\small\color{accent} Main Message}]
        \scriptsize
        GRPO with 7 additive rule-based rewards brings \textbf{all 5 models} (135M--2B) to \textbf{87--99\% Pass@1} on strict JSON generation, single GPU, identical hyperparameters.
      \end{tcolorbox}
      \vspace{0.3em}
      \begin{tcolorbox}[insightbox=sky, title={\small\color{sky} Takeaways}]
        \scriptsize
        \begin{enumerate}
          \setlength\itemsep{0.2em}
          \item \textbf{Standard} optimal for weak models (135M: \textcolor{accent}{+58.67pp})
          \item \textbf{Curriculum} safety net for strong (prevents Gemma regression)
          \item \textbf{Think mode} $<$2pp marginal (small models can't \texttt{<think>})
        \end{enumerate}
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.45\textwidth}
      \begin{tcolorbox}[insightbox=gold, title={\small\color{gold} Limitations}]
        \scriptsize
        \begin{itemize}
          \setlength\itemsep{0.2em}
          \item JSON-only — not directly generalizable
          \item Fully synthetic dataset; real-world gap unknown
          \item Fixed hyperparameters across models
        \end{itemize}
      \end{tcolorbox}
      \vspace{0.3em}
      \begin{tcolorbox}[insightbox=purple, title={\small\color{purple} Future Work}]
        \scriptsize
        \begin{itemize}
          \setlength\itemsep{0.2em}
          \item Extend to YAML, XML, SQL output
          \item Real-world API-call datasets
          \item Reward ablation comparison
          \item Per-model hyperparameter tuning
        \end{itemize}
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Cross-Strategy Insights}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[insightbox=sky]
        \textcolor{sky}{\faBrain}~~\textbf{\color{sky}Think vs.\ No-Think}\\[4pt]
        \scriptsize Small models (135M, 360M, 0.5B) never generate \texttt{<think>} tokens;
        only Gemma and TinyLlama can.
        135M peaks 90\% in Think by exploiting other rewards, not reasoning.
      \end{tcolorbox}
      \vspace{0.5em}
      \begin{tcolorbox}[insightbox=accent]
        \textcolor{accent}{\faLock}~~\textbf{\color{accent}Curriculum = Safety Net}\\[4pt]
        \scriptsize Produces absolute highest scores (TinyLlama \textbf{99.33\%}, Qwen \textbf{98.00\%}).
        Standard causes Gemma to regress.
      \end{tcolorbox}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{tcolorbox}[insightbox=gold]
        \textcolor{gold}{\faChartArea}~~\textbf{\color{gold}Inverted-U Gain Pattern}\\[4pt]
        \scriptsize Smallest models gain most: 135M \textbf{+58pp}.
        Mid-range: +13--26pp.
        Already-strong: +0--8pp (ceiling effect).
      \end{tcolorbox}
      \vspace{0.5em}
      \begin{tcolorbox}[insightbox=warn]
        \textcolor{warn}{\faExclamationTriangle}~~\textbf{\color{warn}Hard Tier Drives Improvement}\\[4pt]
        \scriptsize Baselines: 17--20\% hard pass for 135M.
        Post-GRPO: \textbf{86--99\%} on hard prompts.
        Simple saturates to 100\% in Stage 1.
      \end{tcolorbox}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Reward System — Full Think/No-Think Weight Table}
  \centering
  \begin{tcolorbox}[tablecard]
  \centering
  \scriptsize
  \rowcolors{2}{cardbg}{darkbg!50!cardbg}
  \renewcommand{\arraystretch}{1.45}

  \begin{tabularx}{0.98\linewidth}{@{}l c c c X@{}}
    \rowcolor{cardbg}
    \textbf{Component} & \textbf{Range} & \textbf{\,Think\,} & \textbf{\,NoThink\,} & \textbf{What it measures} \\
    \midrule
    \textcolor{sky}{\faCode}~\texttt{format}     & $[0,1]$      & 0.10 & 0.20 & \texttt{```json```} fence $\to$ 1.0 ~$\mid$~ generic fence $\to$ 0.5 ~$\mid$~ none $\to$ 0.0 \\
    \textcolor{accent}{\faCheck}~\texttt{validity}   & $[0,1]$      & 0.15 & 0.25 & JSON parseability; partial credit by error position (0.20--0.70) \\
    \textcolor{gold}{\faSitemap}~\texttt{schema}     & $[0,1]$      & 0.20 & 0.25 & Keys, types, nesting depth, required fields \\
    \textcolor{purple}{\faBrain}~\texttt{reasoning}  & $[-0.5,1]$   & 0.25 & \alert{0.0}  & \texttt{<think>} block quality; penalises missing/copied \\
    \textcolor{warn}{\faTimes}~\texttt{truncation} & $[-1,1]$     & 0.10 & 0.10 & Unclosed braces / unterminated strings \\
    \textcolor{slate}{\faRedo}~\texttt{repetition} & $[-1,1]$     & 0.10 & 0.10 & Token loops, repeated lines, duplicate blocks \\
    \textcolor{purple}{\faBan}~\texttt{strictness} & $[-1,1]$     & 0.10 & 0.10 & Extra text outside JSON $>$50\% of output \\

  \vspace{0.4em}

  \begin{tcolorbox}[insightbox=sky, width=0.85\textwidth]
    \small\textbf{Design principle} \textcolor{textsecondary}{|}
    Additive composition, no hard gating.
    Gradient is always non-zero; avoids \alert{zero-advantage collapse}.
    When Think is off, reasoning weight is redistributed proportionally.
  \end{tcolorbox}
